```dataviewjs
// Reference date: use current note title if it's a date, else today
const title = dv.current().file.name;

// Strictly accept ISO date titles like YYYY-MM-DD
const isoTitleMatch = /^\d{4}-\d{2}-\d{2}$/.test(title);

// Use title date if valid; otherwise fallback to today
const refDate = isoTitleMatch ? dv.date(title) : dv.date("today");

// Get the date one week ago from the reference date
const target = refDate.minus({ days: 7 });
const targetName = target.toFormat("yyyy-MM-dd");

// Search all daily notes and pick the one whose filename matches the date
const page = dv.pages('"05_DailyNotes"')
  .where(p => p.file.name === targetName)
  .first();

if (!page) {
  dv.paragraph(`No daily note found for: **${targetName}** (reference: **${refDate.toFormat("yyyy-MM-dd")}**)`);
} else {
  // Dataview will usually read inline fields like Weight:: ...
  let w = page.Weight;

  // If you use decimal comma, normalize to a number for safety
  if (typeof w === "string") w = Number(w.replace(",", "."));

  dv.paragraph(`Weight: 1 week ago (**${targetName}**) was **${w ?? "missing"}** kg`);
}
```
