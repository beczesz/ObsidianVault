```dataviewjs
// Reference date: use current note title if it's a date, else today
const title = dv.current().file.name;
const isoTitleMatch = /^\d{4}-\d{2}-\d{2}$/.test(title);
const refDate = isoTitleMatch ? dv.date(title) : dv.date("today");

// Get start of previous week (Monday)
const startOfCurrentWeek = refDate.startOf("week"); // ISO week (Monday)
const startOfLastWeek = startOfCurrentWeek.minus({ weeks: 1 });

// Today's weekday index (0=Monday, 6=Sunday)
const todayIndex = refDate.weekday - 1;

// Collect rows
let rows = [];

for (let i = 0; i < 7; i++) {
    const day = startOfLastWeek.plus({ days: i });
    const dayName = day.toFormat("cccc");
    const dayFile = day.toFormat("yyyy-MM-dd");

    const page = dv.pages('"05_DailyNotes"')
        .where(p => p.file.name === dayFile)
        .first();

    let weight = page?.Weight ?? "missing";

    if (typeof weight === "string")
        weight = Number(weight.replace(",", "."));

    let displayDay = dayName;
    
    // Bold the weekday that corresponds to today
    if (i === todayIndex) {
        displayDay = `**${dayName}**`;
    }

    rows.push([displayDay, dayFile, weight]);
}

dv.table(["Day", "Date", "Weight (kg)"], rows);
```
